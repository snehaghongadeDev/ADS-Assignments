Q1. Trace Bubble Sort Passes 
Problem Description:
Given an array, print the array after each pass of Bubble Sort. 
Example:
Input: [4, 3, 2]
Output: 
Pass 1: [3, 2, 4]
Pass 2: [2, 3, 4]
--------------------Ans-----------------------

public class BubbleSortTrace {
    public static void bubbleSortTrace(int[] arr) {
        int n = arr.length;
        
        // Outer loop for passes
        for (int i = 0; i < n - 1; i++) {
            boolean swapped = false;
            
            System.out.print("Pass " + (i + 1) + ": [");
            // Inner loop for comparisons
            for (int j = 0; j < n - i - 1; j++) {
                if (arr[j] > arr[j + 1]) {
                    // Swap elements
                    int temp = arr[j];
                    arr[j] = arr[j + 1];
                    arr[j + 1] = temp;
                    swapped = true;
                }
                
                // Print element (add comma except for last)
                System.out.print(arr[j]);
                if (j < n - i - 2) System.out.print(", ");
            }
            System.out.println(arr[n - i - 1] + "]");
            
            // Optimization: exit if no swaps
            if (!swapped) break;
        }
    }
    
    public static void main(String[] args) {
        int[] arr1 = {4, 3, 2};
        System.out.println("Input: " + java.util.Arrays.toString(arr1));
        bubbleSortTrace(arr1.clone());  // Use clone to preserve original
        System.out.println();
        
        int[] arr2 = {5, 2, 8, 1, 9};
        System.out.println("Input: " + java.util.Arrays.toString(arr2));
        bubbleSortTrace(arr2.clone());
    }
}

Q2. Insert an Element into a Sorted Array 
Problem Description:
Given a sorted array, insert a new element into its correct position using the logic of Insertion 
Sort. 
Example:
Input: [1, 3, 5, 7], insert 4
Output: [1, 3, 4, 5, 7]
Requirements:
 Do not use extra sorting. 
 Shift elements as needed.
-------------------Ans-------------------

public class InsertIntoSortedArray {
    
    public static int[] insertIntoSorted(int[] sortedArr, int newElement) {
        // Create new array with size +1
        int[] result = new int[sortedArr.length + 1];
        int n = sortedArr.length;
        int i;
        
        // Find correct position and shift elements
        for (i = 0; i < n; i++) {
            if (sortedArr[i] > newElement) {
                break;  // Found insertion point
            }
        }
        
        // Copy elements before insertion point
        for (int j = 0; j < i; j++) {
            result[j] = sortedArr[j];
        }
        
        // Insert new element
        result[i] = newElement;
        
        // Copy remaining elements (shifted right)
        for (int j = i; j < n; j++) {
            result[j + 1] = sortedArr[j];
        }
        
        return result;
    }
    
    public static void main(String[] args) {
        int[] arr1 = {1, 3, 5, 7};
        int[] result1 = insertIntoSorted(arr1, 4);
        System.out.println("Input: " + java.util.Arrays.toString(arr1) + ", insert 4");
        System.out.println("Output: " + java.util.Arrays.toString(result1));  // [1, 3, 4, 5, 7]
        
        System.out.println();
        
        int[] arr2 = {2, 4, 6};
        int[] result2 = insertIntoSorted(arr2, 1);
        System.out.println("Input: " + java.util.Arrays.toString(arr2) + ", insert 1");
        System.out.println("Output: " + java.util.Arrays.toString(result2));  // [1, 2, 4, 6]
        
        System.out.println();
        
        int[] arr3 = {1, 2, 3};
        int[] result3 = insertIntoSorted(arr3, 5);
        System.out.println("Input: " + java.util.Arrays.toString(arr3) + ", insert 5");
        System.out.println("Output: " + java.util.Arrays.toString(result3));  // [1, 2, 3, 5]
    }
}

Q3.Selection Sort in Descending Order 
Problem Description:
Modify the Selection Sort algorithm to sort an array in descending order. 
. 
Example:
Input: [3, 5, 1, 2]
Output: [5, 3, 2, 1]
----------------------Ans-------------

public class SelectionSortDescending {
    
    public static void selectionSortDescending(int[] arr) {
        int n = arr.length;
        
        // One pass for each position
        for (int i = 0; i < n - 1; i++) {
            // Find index of maximum element in unsorted portion
            int maxIndex = i;
            for (int j = i + 1; j < n; j++) {
                if (arr[j] > arr[maxIndex]) {
                    maxIndex = j;  // Found larger element
                }
            }
            
            // Swap maximum with current position
            int temp = arr[i];
            arr[i] = arr[maxIndex];
            arr[maxIndex] = temp;
        }
    }
    
    public static void main(String[] args) {
        int[] arr1 = {3, 5, 1, 2};
        System.out.println("Input: " + java.util.Arrays.toString(arr1));
        selectionSortDescending(arr1);
        System.out.println("Output: " + java.util.Arrays.toString(arr1));  // [5, 3, 2, 1]
        
        System.out.println();
        
        int[] arr2 = {7, 2, 9, 4, 6};
        System.out.println("Input: " + java.util.Arrays.toString(arr2));
        selectionSortDescending(arr2.clone());
        System.out.println("Output: " + java.util.Arrays.toString(arr2));  // [9, 7, 6, 4, 2]
    }
}

Q4.Optimized Bubble Sort 
Problem Description:
Modify the Bubble Sort algorithm to improve its efficiency by stopping early if the array is 
already sorted. 
Requirements:
 If no swaps happen in a pass, terminate early.
----------------Ans----------

public class OptimizedBubbleSort {
    
    public static void optimizedBubbleSort(int[] arr) {
        int n = arr.length;
        
        for (int i = 0; i < n - 1; i++) {
            boolean swapped = false;  // Flag to track swaps in this pass
            
            // Last i elements are already sorted
            for (int j = 0; j < n - i - 1; j++) {
                if (arr[j] > arr[j + 1]) {
                    // Swap elements
                    int temp = arr[j];
                    arr[j] = arr[j + 1];
                    arr[j + 1] = temp;
                    swapped = true;  // Mark that swap occurred
                }
            }
            
            // If no swaps in this pass, array is sorted
            if (!swapped) {
                System.out.println("Early termination: Array already sorted after pass " + (i + 1));
                break;
            }
        }
    }
    
    public static void main(String[] args) {
        // Test Case 1: Already sorted (early termination)
        int[] arr1 = {1, 2, 3, 4, 5};
        System.out.println("Input: " + java.util.Arrays.toString(arr1));
        optimizedBubbleSort(arr1.clone());
        System.out.println("Sorted: " + java.util.Arrays.toString(arr1));
        System.out.println();
        
        // Test Case 2: Reverse sorted (full passes)
        int[] arr2 = {5, 4, 3, 2, 1};
        System.out.println("Input: " + java.util.Arrays.toString(arr2));
        optimizedBubbleSort(arr2.clone());
        System.out.println("Sorted: " + java.util.Arrays.toString(arr2));
        System.out.println();
        
        // Test Case 3: Nearly sorted
        int[] arr3 = {2, 1, 4, 5, 3};
        System.out.println("Input: " + java.util.Arrays.toString(arr3));
        optimizedBubbleSort(arr3.clone());
        System.out.println("Sorted: " + java.util.Arrays.toString(arr3));
    }
}
